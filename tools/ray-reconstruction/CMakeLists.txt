cmake_minimum_required(VERSION 3.18)
project(pwe_ray_rr LANGUAGES CXX)
find_package(Vulkan REQUIRED)
set(NGX_LOADER "" CACHE FILEPATH "v310.9.16 SDK loader: libnvsdk_ngx.a or nvsdk_ngx_s.lib")
if(NOT EXISTS "${NGX_LOADER}")
    message(FATAL_ERROR "Set NGX_LOADER to the official SDK loader library, not the DLSSD runtime")
endif()
add_library(pwe_ray_rr SHARED ngx_bridge.cpp)
target_compile_features(pwe_ray_rr PRIVATE cxx_std_17)
target_include_directories(pwe_ray_rr PRIVATE ../../sdk/include/nvidia ${Vulkan_INCLUDE_DIRS})
target_link_libraries(pwe_ray_rr PRIVATE "${NGX_LOADER}" ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(pwe_ray_rr PRIVATE /W4 /EHsc)
    target_link_libraries(pwe_ray_rr PRIVATE advapi32 version shlwapi)
    # NVIDIA's _s release loader uses the static CRT.
    set_property(TARGET pwe_ray_rr PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
    target_compile_options(pwe_ray_rr PRIVATE -Wall -Wextra -fvisibility=hidden)
    target_link_options(pwe_ray_rr PRIVATE -Wl,--no-undefined)
endif()

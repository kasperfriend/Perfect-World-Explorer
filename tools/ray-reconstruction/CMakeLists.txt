cmake_minimum_required(VERSION 3.18)
project(pwe_ray_rr LANGUAGES CXX)
set(NGX_LOADER "" CACHE FILEPATH "v310.9.16 SDK loader: libnvsdk_ngx.a or nvsdk_ngx_s.lib")
if(WIN32)
    set(loader_name "nvsdk_ngx_s.lib")
    set(loader_sha "7b9a5239c1145af5f645ab472aa066f9ea79db36ac819562312aae69d65f3765")
else()
    set(loader_name "libnvsdk_ngx.a")
    set(loader_sha "d3c50bb0d9624fc35e502a8e929b8fc1e2b505017ebdb84c0d5b627e11a235df")
endif()
get_filename_component(actual_loader "${NGX_LOADER}" NAME)
if(NOT EXISTS "${NGX_LOADER}" OR NOT actual_loader STREQUAL loader_name)
    message(FATAL_ERROR "Set NGX_LOADER to the SDK loader library ${loader_name}, not the DLSSD runtime")
endif()
file(SHA256 "${NGX_LOADER}" actual_sha)
if(NOT actual_sha STREQUAL loader_sha)
    message(FATAL_ERROR "NGX loader library SHA-256 does not match NVIDIA DLSS v310.9.16")
endif()
set(VULKAN_HEADERS "" CACHE PATH "Optional include directory with vulkan/vulkan.h (headers only)")
if(VULKAN_HEADERS)
    if(NOT EXISTS "${VULKAN_HEADERS}/vulkan/vulkan.h")
        message(FATAL_ERROR "VULKAN_HEADERS must contain vulkan/vulkan.h")
    endif()
    set(Vulkan_INCLUDE_DIRS "${VULKAN_HEADERS}")
else()
    find_package(Vulkan REQUIRED)
endif()
add_library(pwe_ray_rr SHARED ngx_bridge.cpp)
target_compile_features(pwe_ray_rr PRIVATE cxx_std_17)
target_include_directories(pwe_ray_rr PRIVATE ../../sdk/include/nvidia ${Vulkan_INCLUDE_DIRS})
target_link_libraries(pwe_ray_rr PRIVATE "${NGX_LOADER}" ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(pwe_ray_rr PRIVATE /W4 /EHsc)
    target_link_libraries(pwe_ray_rr PRIVATE advapi32 version shlwapi)
    # NVIDIA's _s release loader uses the static CRT.
    set_property(TARGET pwe_ray_rr PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
    target_compile_options(pwe_ray_rr PRIVATE -Wall -Wextra -fvisibility=hidden)
    target_link_options(pwe_ray_rr PRIVATE -Wl,--no-undefined)
endif()
